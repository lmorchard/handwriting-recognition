import torch
from craft import CRAFT
from collections import OrderedDict

def load_craft_model():
    """Load CRAFT with updated PyTorch compatibility."""
    net = CRAFT()
    
    # Load state dict safely with PyTorch 2.x
    state_dict = torch.load("./craft_mlt_25k.pth", map_location="cpu")

    # Fix for PyTorch versions that add 'module.' to state_dict keys
    new_state_dict = OrderedDict()
    for k, v in state_dict.items():
        new_key = k.replace("module.", "")  # Remove 'module.' prefix if present
        new_state_dict[new_key] = v

    net.load_state_dict(new_state_dict, strict=False)
    net.eval()
    
    return net

import numpy as np

def merge_word_boxes_into_lines(df, y_threshold=10, x_spacing_threshold=15):
    """
    Groups word-level bounding boxes into line-level bounding boxes.
    
    Args:
        df (DataFrame): DataFrame containing Tesseract's word bounding boxes.
        y_threshold (int): Maximum Y-axis difference for words to be considered in the same line.
        x_spacing_threshold (int): Maximum gap between words in a line before splitting.

    Returns:
        List of merged line bounding boxes [(x1, y1, x2, y2), ...]
    """
    if df.empty:
        print("⚠️ No words detected.")
        return []

    # Sort by y-coordinate first, then x-coordinate (top to bottom, then left to right)
    df = df.sort_values(by=["top", "left"])

    lines = []  # Stores merged bounding boxes for each line
    current_line = []  # Holds words that belong to the same line

    for i, row in df.iterrows():
        x, y, w, h = row["left"], row["top"], row["width"], row["height"]
        word_box = (x, y, x + w, y + h)

        if not current_line:
            current_line.append(word_box)
            continue

        # Check if the new word belongs to the same line (Y difference is small)
        last_box = current_line[-1]
        last_x2, last_y1, last_y2 = last_box[2], last_box[1], last_box[3]

        if abs(y - last_y1) <= y_threshold and abs(y + h - last_y2) <= y_threshold:
            # Same line: Check horizontal distance
            if x - last_x2 <= x_spacing_threshold:
                current_line.append(word_box)
                continue

        # If not part of the same line, finalize the previous line
        x1 = min([b[0] for b in current_line])
        y1 = min([b[1] for b in current_line])
        x2 = max([b[2] for b in current_line])
        y2 = max([b[3] for b in current_line])
        lines.append((x1, y1, x2, y2))

        # Start a new line with the current word
        current_line = [word_box]

    # Add the last detected line
    if current_line:
        x1 = min([b[0] for b in current_line])
        y1 = min([b[1] for b in current_line])
        x2 = max([b[2] for b in current_line])
        y2 = max([b[3] for b in current_line])
        lines.append((x1, y1, x2, y2))

    return lines

import numpy as np
from sklearn.cluster import DBSCAN

def merge_word_boxes_smart(df, image_width, image_height, y_scale=0.01, x_scale=0.003):
    """
    Groups word-level bounding boxes into line-level bounding boxes using clustering.
    
    Args:
        df (DataFrame): DataFrame with word bounding boxes.
        image_width (int): Width of the image.
        image_height (int): Height of the image.
        y_scale (float): Y threshold as % of image height (default: 1%).
        x_scale (float): X threshold as % of image width (default: 0.3%).

    Returns:
        List of merged line bounding boxes [(x1, y1, x2, y2), ...]
    """
    if df.empty:
        print("⚠️ No words detected.")
        return []

    # Convert to NumPy array
    word_boxes = df[['left', 'top', 'width', 'height']].to_numpy()
    
    # Compute center of each box (for clustering)
    centers = np.array([[x + w / 2, y + h / 2] for x, y, w, h in word_boxes])

    # Dynamic Y threshold (based on image size)
    y_threshold = int(image_height * y_scale)
    x_threshold = int(image_width * x_scale)

    # DBSCAN clustering based on Y-position (vertical grouping)
    clustering = DBSCAN(eps=y_threshold, min_samples=1, metric='euclidean').fit(centers)

    # Assign cluster labels to words
    df['line_id'] = clustering.labels_

    # Merge bounding boxes within each line
    lines = []
    for line_id in sorted(df['line_id'].unique()):
        line_words = df[df['line_id'] == line_id]

        x1 = min(line_words['left'])
        y1 = min(line_words['top'])
        x2 = max(line_words['left'] + line_words['width'])
        y2 = max(line_words['top'] + line_words['height'])

        lines.append((x1, y1, x2, y2))

    return lines

import numpy as np
import cv2

def detect_text_lines_using_histogram(image, df, binarize=True, min_line_height=5, min_word_gap=50):
    """
    Detects lines of text using histogram projection + connected components.

    Args:
        image (np.array): The input image.
        df (DataFrame): Word bounding boxes from Tesseract.
        binarize (bool): Whether to threshold the image before analysis.
        min_line_height (int): Minimum pixel height of a detected text line.
        min_word_gap (int): Maximum allowed gap between words to merge them into a line.

    Returns:
        List of merged line bounding boxes [(x1, y1, x2, y2), ...]
    """
    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)

    # Step 1: Apply thresholding for cleaner histograms
    if binarize:
        _, binary = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)
    else:
        binary = gray

    # Step 2: Calculate vertical histogram projection
    hist = np.sum(binary, axis=1)  # Sum pixel values along columns (detects text density)

    # Step 3: Detect text regions based on histogram peaks
    in_text_region = False
    line_positions = []

    for y, value in enumerate(hist):
        if value > min_line_height and not in_text_region:
            in_text_region = True
            start_y = y
        elif value <= min_line_height and in_text_region:
            in_text_region = False
            line_positions.append((start_y, y))

    # Step 4: Merge detected words into line bounding boxes
    line_boxes = []
    for (y_start, y_end) in line_positions:
        words_in_line = df[(df["top"] >= y_start) & (df["top"] + df["height"] <= y_end)].sort_values("left")

        if words_in_line.empty:
            continue

        # Merge horizontally adjacent words
        x1 = words_in_line["left"].min()
        x2 = words_in_line["left"].max() + words_in_line["width"].max()
        y1 = y_start
        y2 = y_end

        # Handle large gaps between words (if spacing is large, split lines)
        word_gaps = np.diff(words_in_line["left"].to_numpy())
        if any(word_gaps > min_word_gap):
            # Split into multiple sub-lines based on gaps
            split_indexes = np.where(word_gaps > min_word_gap)[0] + 1
            sub_lines = np.array_split(words_in_line, split_indexes)
            for sub_line in sub_lines:
                if len(sub_line) > 0:
                    x1_sub = sub_line["left"].min()
                    x2_sub = sub_line["left"].max() + sub_line["width"].max()
                    line_boxes.append((x1_sub, y1, x2_sub, y2))
        else:
            line_boxes.append((x1, y1, x2, y2))

    return line_boxes

import numpy as np
import cv2

def cluster_lines_by_y_histogram(df, image, y_tolerance_factor=0.02):
    """
    Groups word bounding boxes into line-level bounding boxes using Y-axis histogram clustering.

    Args:
        df (DataFrame): Word bounding boxes from Tesseract.
        image (np.array): The original image.
        y_tolerance_factor (float): How much Y-variation is allowed in a single line (as % of image height).

    Returns:
        List of merged line bounding boxes [(x1, y1, x2, y2), ...]
    """
    if df.empty:
        print("⚠️ No words detected.")
        return []

    # Convert word bounding boxes to a binary mask
    height, width = image.shape[:2]
    binary_mask = np.zeros((height, width), dtype=np.uint8)

    for _, row in df.iterrows():
        x1, y1, x2, y2 = row["left"], row["top"], row["left"] + row["width"], row["top"] + row["height"]
        binary_mask[y1:y2, x1:x2] = 255  # Mark text areas

    # Compute vertical projection (sum of pixels along each row)
    y_histogram = np.sum(binary_mask, axis=1)

    # Detect peaks (text lines)
    line_positions = []
    in_text_region = False
    y_threshold = int(height * y_tolerance_factor)  # Dynamic Y tolerance

    for y, value in enumerate(y_histogram):
        if value > 0 and not in_text_region:
            in_text_region = True
            start_y = y
        elif value == 0 and in_text_region:
            in_text_region = False
            if y - start_y > y_threshold:
                line_positions.append((start_y, y))

    # Cluster words into detected Y-ranges
    lines = []
    for (y_start, y_end) in line_positions:
        words_in_line = df[(df["top"] >= y_start) & (df["top"] + df["height"] <= y_end)]

        if words_in_line.empty:
            continue

        # Merge all words in the Y-region into a single bounding box
        x1 = words_in_line["left"].min()
        x2 = words_in_line["left"].max() + words_in_line["width"].max()
        y1 = y_start
        y2 = y_end

        lines.append((x1, y1, x2, y2))

    return lines

    - Prevents cutting off descenders or ascenders by dynamically setting Y2.

        min_spacing_factor (float): Minimum spacing between lines (as % of image height).

    # Get image dimensions

        binary_mask[y1:y2, x1:x2] = 255  # Mark text areas

    # Detect peaks (actual text lines) in the Y-histogram

        df[(df["top"] <= peak_y) & (df["top"] + df["height"] >= peak_y)]["height"].max() for peak_y in peaks

import cv2
import numpy as np
import matplotlib.pyplot as plt

def detect_words_and_lines(image, line_kernel_size=100, word_kernel_size=20):
    """
    Detects text lines and words purely using whitespace analysis (no OCR).

    Args:
        image (np.array): Input grayscale or RGB image.
        line_kernel_size (int): Size of the horizontal kernel for line detection.
        word_kernel_size (int): Size of the vertical kernel for word separation.

    Returns:
        line_boxes (list of tuples): Bounding boxes of detected lines [(x1, y1, x2, y2), ...].
        word_boxes (list of tuples): Bounding boxes of detected words [(x1, y1, x2, y2), ...].
    """
    # Convert to grayscale if needed
    if len(image.shape) == 3:
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    else:
        gray = image.copy()

    # Apply adaptive thresholding (handles varying handwriting intensity)
    binary = cv2.adaptiveThreshold(gray, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C, cv2.THRESH_BINARY_INV, 31, 2)

    ### **Step 1: Detect Lines Using Horizontal Projection**
    line_kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (line_kernel_size, 1))  # Large horizontal kernel
    detected_lines = cv2.morphologyEx(binary, cv2.MORPH_CLOSE, line_kernel, iterations=2)

    # Find contours for text lines
    line_contours, _ = cv2.findContours(detected_lines, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    line_boxes = [cv2.boundingRect(c) for c in line_contours]

    ### **Step 2: Detect Words Using Vertical Projection**
    word_kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (word_kernel_size, word_kernel_size))  # Small square kernel
    detected_words = cv2.morphologyEx(binary, cv2.MORPH_CLOSE, word_kernel, iterations=1)

    # Find contours for words
    word_contours, _ = cv2.findContours(detected_words, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    word_boxes = [cv2.boundingRect(c) for c in word_contours]

    return line_boxes, word_boxes

import torch

JOURNAL_PAGE_PATH = "journal_page.png"  # The full journal page PNG
OUTPUT_IMAGE_PATH = "annotated_journal.png"  # Where the annotated image will be saved

# **Step 1: Load Fine-Tuned TrOCR Model**
device = "cuda" if torch.cuda.is_available() else "cpu"
processor = TrOCRProcessor.from_pretrained(MODEL_PATH)
model = VisionEncoderDecoderModel.from_pretrained(MODEL_PATH).to(device)

# **Step 2: Detect Lines Using Whitespace-Based Method**
def detect_lines_by_whitespace(image, line_kernel_size=100):
    """Detects text lines using horizontal whitespace projection."""
    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    binary = cv2.adaptiveThreshold(gray, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C, cv2.THRESH_BINARY_INV, 31, 2)
    
    # Detect lines
    line_kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (line_kernel_size, 1))
    detected_lines = cv2.morphologyEx(binary, cv2.MORPH_CLOSE, line_kernel, iterations=2)
    
    # Find contours
    contours, _ = cv2.findContours(detected_lines, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    line_boxes = [cv2.boundingRect(c) for c in contours]

    # Convert to full-width bounding boxes
    height, width = image.shape[:2]
    full_width_boxes = [(0, y, width, y + h) for (x, y, w, h) in line_boxes]
    
    return sorted(full_width_boxes, key=lambda box: box[1])  # Sort lines top-to-bottom

# Load the journal page
image = cv2.imread(JOURNAL_PAGE_PATH)
line_boxes = detect_lines_by_whitespace(image)

# **Step 3: Run OCR on Each Line**
def recognize_text_from_line(image, line_box):
    """Runs OCR on a cropped line from the journal page."""
    x1, y1, x2, y2 = line_box
    cropped_line = image[y1:y2, x1:x2]
    cropped_pil = Image.fromarray(cv2.cvtColor(cropped_line, cv2.COLOR_BGR2RGB))

    # Run TrOCR
    pixel_values = processor(cropped_pil, return_tensors="pt").pixel_values.to(device)
    with torch.no_grad():
        generated_ids = model.generate(pixel_values)

    return processor.batch_decode(generated_ids, skip_special_tokens=True)[0]

# **Step 4: Annotate the Image with Recognized Text**
annotated_image = image.copy()

for line_box in line_boxes:
    text = recognize_text_from_line(image, line_box)

    # Draw bounding box
    x1, y1, x2, y2 = line_box
    cv2.rectangle(annotated_image, (x1, y1), (x2, y2), (0, 255, 0), 2)

    # Draw recognized text
    font = cv2.FONT_HERSHEY_SIMPLEX
    font_scale = max(0.5, (y2 - y1) / 50)
    thickness = 2
    text_x, text_y = x1 + 10, y1 + 30
    cv2.putText(annotated_image, text, (text_x, text_y), font, font_scale, (0, 0, 255), thickness, cv2.LINE_AA)

# **Step 5: Save & Display the Annotated Image**
cv2.imwrite(OUTPUT_IMAGE_PATH, annotated_image)

# Show the annotated image

plt.imshow(cv2.cvtColor(annotated_image, cv2.COLOR_BGR2RGB))
plt.axis("off")
plt.show()

print(f"✅ Annotated image saved as '{OUTPUT_IMAGE_PATH}'")

    """Detects text lines using horizontal whitespace projection."""

    full_width_boxes = [(0, y, width, y + h) for (x, y, w, h) in line_boxes]
