image_path = "/content/my_journal_page_thresholded.png"

east_model = "frozen_east_text_detection.pb"

# Define parameters
newW, newH = (320, 320)
rW = W / float(newW)
rH = H / float(newH)

# Resize image
blob = cv2.dnn.blobFromImage(image, 1.0, (newW, newH),
                             (123.68, 116.78, 103.94), swapRB=True, crop=False)

# Load the EAST model
net = cv2.dnn.readNet(east_model)

3️⃣ Perform Text Detection

# Forward pass through EAST model
net.setInput(blob)
(scores, geometry) = net.forward(["feature_fusion/Conv_7/Sigmoid", "feature_fusion/concat_3"])

# Extract bounding boxes
def decode_predictions(scores, geometry, min_confidence=0.5):
    rects, confidences = [], []
    (rows, cols) = scores.shape[2:4]
    
    for y in range(rows):
        scores_data = scores[0, 0, y]
        x_data0, x_data1, x_data2, x_data3, angles_data = [geometry[0, i, y] for i in range(5)]
        
        for x in range(cols):
            if scores_data[x] < min_confidence:
                continue
            
            # Compute bounding box offsets
            offset_x, offset_y = x * 4.0, y * 4.0
            angle = angles_data[x]
            cos, sin = np.cos(angle), np.sin(angle)
            
            h, w = x_data0[x] + x_data2[x], x_data1[x] + x_data3[x]
            end_x = int(offset_x + (cos * x_data1[x]) + (sin * x_data2[x]))
            end_y = int(offset_y - (sin * x_data1[x]) + (cos * x_data2[x]))
            start_x, start_y = int(end_x - w), int(end_y - h)
            
            rects.append((start_x, start_y, end_x, end_y))
            confidences.append(scores_data[x])

    return rects, confidences

# Decode predictions
from imutils.object_detection import non_max_suppression
boxes, confidences = decode_predictions(scores, geometry)
boxes = non_max_suppression(np.array(boxes), probs=confidences)
